async function showLocalNotification(title,body){
  if(!('Notification'in window)){toast('Notifications is browser me supported nahi hai');return false}
  let p=Notification.permission;
  if(p==='default'){
    try{p=await Notification.requestPermission()}catch(e){console.warn(e)}
  }
  if(p!=='granted'){
    toast('Notification blocked hai • Chrome/App settings me Allow karo');
    renderNotificationStatus();
    return false
  }
  settings.notifications=true;
  safeStorage.setItem(SETTINGS_KEY,JSON.stringify(settings));
  try{
    const tag='cfm-event-'+Date.now();
    if('serviceWorker'in navigator&&location.protocol.startsWith('http')){
      const reg=await navigator.serviceWorker.ready;
      await reg.showNotification(title,{body,icon:'icons/icon-192.png',badge:'icons/icon-192.png',tag,renotify:true});
    }else new Notification(title,{body,tag});
    renderNotificationStatus();
    return true
  }catch(e){
    console.warn(e);
    toast('Notification show nahi hua • permission/settings check karo');
    return false
  }
}

function loadExcelMirrorData(){
  if(excelMirrorData)return excelMirrorData;
  try{
    const node=document.getElementById('excelMirrorSource');
    if(!node)throw new Error('Mirror source missing');
    const raw=node.textContent||node.innerText||'';
    if(!raw.trim())throw new Error('Mirror source empty');
    const parsed=JSON.parse(raw);
    if(!parsed||!Array.isArray(parsed.sheets)||parsed.sheets.length!==78)throw new Error('Mirror source incomplete');
    excelMirrorData=parsed;
    return excelMirrorData;
  }catch(e){console.warn('CFM mirror load:',e);toast('Excel mirror load error — reopen file once');return null}
}

window.renderExcelMirror=function(){
  const d=loadExcelMirrorData();if(!d)return;
  const name=$('excelMirrorSheet')?.value||d.sheets[0]?.name,s=d.sheets.find(x=>x.name===name);if(!s)return;
  const q=($('excelMirrorSearch')?.value||'').trim().toLowerCase();
  let rows=s.rows;
  if(q)rows=rows.filter(r=>r[1].some(c=>String(c[1]??'').toLowerCase().includes(q)));
  const pages=Math.max(1,Math.ceil(rows.length/EXCEL_MIRROR_PAGE_SIZE));excelMirrorPage=Math.max(0,Math.min(excelMirrorPage,pages-1));
  const start=excelMirrorPage*EXCEL_MIRROR_PAGE_SIZE,part=rows.slice(start,start+EXCEL_MIRROR_PAGE_SIZE);
  if($('excelMirrorMeta'))$('excelMirrorMeta').innerHTML=`<span class="chip green">Sheet ${s.index}/78</span><span class="chip">${esc(s.name)}</span><span class="chip">${rows.length.toLocaleString()} matched rows</span><span class="chip">${s.nonEmptyCells.toLocaleString()} cells</span><span class="chip">${s.maxCol} used columns</span>`;
  if($('excelMirrorPageLabel'))$('excelMirrorPageLabel').textContent=`Page ${excelMirrorPage+1} / ${pages}`;
  const box=$('excelMirrorGrid');if(!box)return;
  if(!part.length){box.innerHTML='<div class="empty">No matching rows</div>';return}
  const head=`<thead><tr><th>#</th>${Array.from({length:s.maxCol},(_,i)=>`<th>${colLetter(i)}</th>`).join('')}</tr></thead>`;

    return `<tr><td>${r[0]}</td>${Array.from({length:s.maxCol},(_,i)=>`<td>${esc(map.get(i)??'')}</td>`).join('')}</tr>`;
  }).join('');
  box.innerHTML=`<table>${head}<tbody>${body}</tbody></table>`;
}
