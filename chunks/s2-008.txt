function genericUpdateNotification(prevRaw,nextRaw){
  try{
    const prev=prevRaw?JSON.parse(prevRaw):{}, next=nextRaw?JSON.parse(nextRaw):{};
    const changed=[];
    const same=(a,b)=>JSON.stringify(a||[])===JSON.stringify(b||[]);
    if(!same(prev.pending,next.pending))changed.push('Pending');
    if(!same(prev.cutting,next.cutting))changed.push('Cutting');
    if(!same(prev.store,next.store))changed.push('Store');
    if(!same(prev.departmentWork,next.departmentWork))changed.push('Person Work');
    if(!same(prev.measurements,next.measurements))changed.push('Measurement');
    if(!same(prev.srlTracker,next.srlTracker))changed.push('SRL Tracker');
    if(!same(prev.issueRegister,next.issueRegister))changed.push('Issue Register');
    if(!same(prev.srlOpenClose,next.srlOpenClose))changed.push('SRL Status');
    if(!same(prev.productMaster,next.productMaster))changed.push('Product Master');
    const latest=(arr=[])=>arr.slice().sort((a,b)=>String(b.updatedAt||'').localeCompare(String(a.updatedAt||'')))[0]||{};
    let detail='';
    if(changed.length===1&&changed[0]==='Pending'){
      const r=latest(next.pending);detail=[r.workNo,r.party,r.description2||r.description].filter(Boolean).join(' • ');
    }else if(changed.length===1&&changed[0]==='Cutting'){
      const r=latest(next.cutting);detail=[r.workNo,r.cutQty!==''?('Qty '+fmt(r.cutQty)):'',r.srlNo?('SRL '+r.srlNo):''].filter(Boolean).join(' • ');
    }else if(changed.length===1&&changed[0]==='Store'){
      const r=latest(next.store);detail=[r.srlNo,r.quality,r.openingMtr!==''?('MTR '+fmt(r.openingMtr)):''].filter(Boolean).join(' • ');
    }else if(changed.length===1&&changed[0]==='Person Work'){
      const r=latest(next.departmentWork);detail=[r.person,r.process,r.description2||r.description].filter(Boolean).join(' • ');
    }else if(changed.length===1&&changed[0]==='Measurement'){
      const r=latest(next.measurements);detail=[r.workNo,r.eagleCode,r.finalMeasurement||r.measurement3||r.measurement2||r.measurement1].filter(Boolean).join(' • ');
    }
    return {
      title:changed.length?('🔔 '+changed.join(' + ')+' Updated'):'🔔 CFM Data Updated',
      body:[detail,deviceLabel()].filter(Boolean).join(' • ')
    };
  }catch(e){
    return {title:'🔔 CFM Data Updated',body:deviceLabel()};
  }
}

function cuttingUsedMtr(r){
  if(!r || r.wastage || normalize(r.srlNo)==='W') return 0;
  const q=num(r.cutQty), l=num(r.length), h=num(r.height), layer=Math.max(1,num(r.layer)||1);
  const s=getStore(r.srlNo);
  const width=num(r.plannedWidth)||num(s?.width);
  if(!(q>0&&l>0&&h>0&&width>0)) return 0;
  return (l*h*q*layer/width/39.37);
}

function renderPending(){
  const q=($('pendingSearch').value||'').toLowerCase(),mode=$('pendingStatus').value;
  let rows=state.pending.filter(r=>JSON.stringify(r).toLowerCase().includes(q));
  if(mode==='OPEN')rows=rows.filter(orderOpen);
  if(mode==='CUTENTRY')rows=rows.filter(r=>!isArchived(r)&&isCutEntryPending(r));
  if(mode==='ALL')rows=rows.filter(r=>!isArchived(r));
  if(mode==='COMPLETE')rows=rows.filter(r=>!isArchived(r)&&!isCutEntryPending(r)&&hasOrderQty(r)&&pendingQty(r)<=0);
  if(mode==='ARCHIVED')rows=rows.filter(isArchived);
  rows.sort((a,b)=>String(b.workNo||'').localeCompare(String(a.workNo||''),undefined,{numeric:true}));
  $('pendingList').innerHTML=rows.map(r=>pendingCard(r)).join('')||'<div class="empty">No matching orders</div>';
}

window.markCutEntryPending=function(i){
  const r=state.pending[i];if(!r||isArchived(r))return;
  if(!hasOrderQty(r))return toast('Order Qty pehle enter karo');
  if(!confirm((r.workNo||'This work')+' ka physical cutting complete ho gaya hai?\n\nYe normal Pending list se hatega aur Cut Entry Pending me rahega jab tak cutting entries complete nahi hoti.'))return;
  r.cutEntryPending=true;
  r.cutEntryPendingAt=new Date().toISOString();
  r.cutEntryPendingBy=deviceLabel();
  r.planDate='';
  stampRecord(r);
  setNextUpdateNotification('✂ Cutting Done • Entry Pending',[r.workNo||'Work',r.party||'','Entry balance '+fmt(pendingQty(r))].filter(Boolean).join(' • '));
  save();
  toast('Pending se hata diya • Cut Entry Pending me safe hai');
}
window.restoreCutEntryPending=function(i){
  const r=state.pending[i];if(!r)return;
  r.cutEntryPending=false;r.cutEntryPendingAt='';r.cutEntryPendingBy='';
  stampRecord(r);save();toast('Work wapas normal Pending me aa gaya');
}

window.archivePending=function(i){
  const r=state.pending[i];if(!r)return;
  if(!confirm(`Archive ${r.workNo||'this order'}?\n\nPending se hat jayega, history me safe rahega aur baad me Restore kar sakte ho.`))return;
  r.archived=true;r.archivedAt=new Date().toISOString();r.archivedBy=deviceLabel();r.planDate='';r.status='ARCHIVED';stampRecord(r);save();toast('Order archived • Restore available');
}

window.restorePending=function(i){
  const r=state.pending[i];if(!r)return;
  r.archived=false;r.archivedAt='';r.archivedBy='';r.status=!hasOrderQty(r)?'QTY REQUIRED':(pendingQty(r)<=0?'COMPLETED':'PENDING');stampRecord(r);save();toast('Order restored to Pending');
}
function workHasLinkedActivity(r){
  const work=normalize(r?.workNo);
  const cutting=(state.cutting||[]).filter(c=>normalize(c.workNo)===work);
  const hasCut=cutting.length>0 || num(r?.cutQty)>0;
  const hasDocs=['billNo','billDate','invoiceValue','challanNo','challanDate','dispatchStatus','dispatchDate','dispatchQty','transporter','vehicleNo','lrNo','lrDate','ewayBillNo','ewayBillDate'].some(k=>String(r?.[k]??'').trim());
  return {hasCut,hasDocs,cuttingCount:cutting.length};
}
