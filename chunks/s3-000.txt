function productLabel(p){return `${p.eagleCode||''} | ${p.description||''}${p.itemCode?` | ${p.itemCode}`:''}`}
function previousOrderCandidates(party=''){
  const p=normalize(party);
  const out=[],seen=new Set();

  // 1) Complete old Excel order history. Keep every old Work No./detail;
  // do not collapse repeated products from different old work orders.
  let history=(FULL_OLD_ORDERS||[]).filter(x=>x.workNo || x.description);
  let histParty=p?history.filter(x=>normalize(x.party)===p):history;
  if(!histParty.length && p){
    // If selected spelling has no exact old match, retain all history so user can still search.
    histParty=history;
  }
  histParty.forEach(x=>{
    const key=`OLD|${normalize(x.workNo)}|${normalize(x.party)}|${normalize(x.description)}`;
    if(seen.has(key))return;seen.add(key);
    out.push({...x,source:'OLD EXCEL'});
  });

  // 2) Current app/cloud orders that may be newer than the source workbook.
  const current=(state.pending||[]).filter(r=>!isArchived(r) && (r.workNo||r.description));
  let curParty=p?current.filter(r=>normalize(r.party)===p):current;
  if(!curParty.length && p)curParty=current;
  curParty.slice().reverse().forEach(r=>{
    const key=`CUR|${normalize(r.workNo)}|${normalize(r.party)}|${normalize(r.description)}`;
    if(seen.has(key))return;seen.add(key);
    out.push({
      source:'CURRENT ERP',
      workNo:r.workNo||'',
      party:r.party||'',
      description:r.description||'',
      eagleCode:r.eagleCode||'',
      itemCode:r.itemCode||'',
      cloth:r.cloth||'',
      width:r.width||'',
      poNo:r.poNo||'',
      orderDate:r.orderDate||'',
      orderQty:r.orderQty??''
    });
  });

  // 3) Product Master stays available for totally new/repeat product selection.
  const products=(state.productMaster||[]).filter(x=>x.description);
  products.forEach(x=>{
    const key=`PM|${normalize(x.description)}|${normalize(x.eagleCode)}|${normalize(x.itemCode)}`;
    if(seen.has(key))return;seen.add(key);
    out.push({
      source:'PRODUCT MASTER',
      workNo:'',
      party:'',
      description:x.description||'',
      eagleCode:x.eagleCode||'',
      itemCode:x.itemCode||'',
      cloth:x.standardCloth||'',
      width:x.standardWidth||'',
      poNo:'',
      orderDate:'',
      orderQty:''
    });
  });
  return out;
}

async function compressImage(file){
  if(!file.type.startsWith('image/') || file.size<900000)return file;
  try{
    const bmp=await createImageBitmap(file), max=1600, scale=Math.min(1,max/Math.max(bmp.width,bmp.height));
    const c=document.createElement('canvas');c.width=Math.round(bmp.width*scale);c.height=Math.round(bmp.height*scale);
    c.getContext('2d').drawImage(bmp,0,0,c.width,c.height);
    return await new Promise(res=>c.toBlob(b=>res(b||file),'image/jpeg',0.78));
  }catch(e){return file}
}
function attachSafeName(s){
  return String(s||'file').replace(/[^\w.\-]+/g,'_').replace(/_+/g,'_').slice(0,120);
}
function storageObjectURL(path){
  return CLOUD_URL+'/storage/v1/object/'+ATTACH_BUCKET+'/'+String(path).split('/').map(encodeURIComponent).join('/');
}
function storageDownloadURL(path){
  return CLOUD_URL+'/storage/v1/object/authenticated/'+ATTACH_BUCKET+'/'+String(path).split('/').map(encodeURIComponent).join('/');
}
function attachmentApiHeaders(extra={}){
  return {...cloudHeaders(),...extra};
}
async function cloudAttachmentList(workNo){
  const u=CLOUD_URL+'/rest/v1/cfm_attachments?work_no=eq.'+encodeURIComponent(workNo)+'&select=id,work_no,file_name,mime_type,size_bytes,storage_path,created_at,uploaded_by&order=created_at.desc';
  const r=await fetch(u,{headers:attachmentApiHeaders(),cache:'no-store'});
  if(!r.ok)throw new Error('Attachment list '+r.status);
  const a=await r.json();
  return (a||[]).map(x=>({...x,source:'cloud'}));
}
async function cloudAttachmentUpload(workNo,file,name){
  const path=attachSafeName(workNo)+'/'+Date.now()+'-'+Math.random().toString(36).slice(2,9)+'-'+attachSafeName(name);
  const up=await fetch(storageObjectURL(path),{
    method:'POST',
    headers:attachmentApiHeaders({'Content-Type':file.type||'application/octet-stream','x-upsert':'false'}),
    body:file
  });
  if(!up.ok)throw new Error('Cloud file upload '+up.status+' '+await up.text().catch(()=>''));
  const meta={work_no:workNo,file_name:name,mime_type:file.type||'application/octet-stream',size_bytes:file.size,storage_path:path,uploaded_by:deviceLabel()};
  const mr=await fetch(CLOUD_URL+'/rest/v1/cfm_attachments',{
    method:'POST',
    headers:attachmentApiHeaders({'Prefer':'return=representation'}),
    body:JSON.stringify(meta)
  });
  if(!mr.ok)throw new Error('Cloud metadata '+mr.status+' '+await mr.text().catch(()=>''));
  const rows=await mr.json();
  return rows?.[0]||meta;
}
async function cloudAttachmentBlob(path){
  const r=await fetch(storageDownloadURL(path),{headers:attachmentApiHeaders({'Content-Type':'application/octet-stream'}),cache:'no-store'});
  if(!r.ok)throw new Error('Cloud download '+r.status);
  return await r.blob();
}
async function cloudAttachmentDelete(rec){
  let storageOk=false;
  try{
    const dr=await fetch(storageObjectURL(rec.storage_path),{method:'DELETE',headers:attachmentApiHeaders()});
    storageOk=dr.ok;
  }catch(e){}
  const u=CLOUD_URL+'/rest/v1/cfm_attachments?id=eq.'+encodeURIComponent(rec.id);
  const mr=await fetch(u,{method:'DELETE',headers:attachmentApiHeaders()});
  if(!mr.ok)throw new Error('Cloud metadata delete '+mr.status);
  return storageOk;
}

window.handleAttachmentFiles=async function(files){
  if(!attachmentWorkNo||!files?.length)return;
  let cloudCount=0,localCount=0;
  try{
    for(const f of [...files]){
      if(f.size>15*1024*1024){toast(`${f.name}: 15 MB se bada hai`);continue}
      const data=f.type.startsWith('image/')?await compressImage(f):f;
      const name=(data!==f && data.type==='image/jpeg')?f.name.replace(/\.[^.]+$/, '')+'.jpg':f.name;
      let savedCloud=false;
      if(navigator.onLine){
        try{
          await cloudAttachmentUpload(attachmentWorkNo,data,name);
          savedCloud=true;cloudCount++;
        }catch(e){console.warn('Cloud attachment failed, using local fallback',e)}
      }
      if(!savedCloud){
        await attachPut({id:`${attachmentWorkNo}-${Date.now()}-${Math.random().toString(36).slice(2)}`,workNo:attachmentWorkNo,name,type:data.type||f.type||'application/octet-stream',size:data.size,createdAt:new Date().toISOString(),data,source:'local'});
        localCount++;
      }
    }
    if(cloudCount&&localCount)toast(`${cloudCount} cloud • ${localCount} local fallback`);
    else if(cloudCount)toast('☁️ Attachment synced • dusre mobile me dikhega');
    else toast('Offline/local save • cloud sync nahi hua');
    await renderAttachmentList();
  }catch(e){console.error(e);toast('Attachment save nahi hua')}
}
