 window.openWorkTrack=function(w){
   const p=getPending(w),cuts=(state.cutting||[]).filter(x=>norm(x.workNo)===norm(w)),dept=(state.departmentWork||[]).filter(x=>norm(x.workNo)===norm(w)),issues=(state.issueRegister||[]).filter(x=>norm(x.workNo)===norm(w));
   const srls=[...new Set(cuts.map(x=>x.srlNo).concat(issues.map(x=>x.srlNo)).filter(x=>x&&norm(x)!=='W'))];
   let status=p?.status||'OPEN';
   const proc=dept.map(x=>norm(x.process||x.department||''));
   if(p?.dispatchStatus||p?.dispatchDate)status='DISPATCHED';
   else if(p?.challanNo||p?.challanStatus==='DONE')status='READY / CHALLAN';
   else if(proc.some(x=>x.includes('PACK')))status='PACKING';
   else if(proc.some(x=>x.includes('FINISH')||x.includes('GUN')||x.includes('THREAD')))status='FINISHING';
   else if(proc.some(x=>x.includes('STITCH')))status='STITCHING';
   else if(p&&hasOrderQty(p)&&pendingQty(p)<=0)status='CUTTING COMPLETE';
   else if(cuts.length)status='CUTTING IN PROGRESS';
